cmake_minimum_required(VERSION 3.22)
project(xtw5-linux VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets Concurrent)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
add_library(xtwcore src/chip.cpp src/protocol.cpp src/transport.cpp src/programmer.cpp src/buffer.cpp)
target_include_directories(xtwcore PUBLIC src)
target_link_libraries(xtwcore PUBLIC Qt6::Core OpenSSL::Crypto PkgConfig::LIBUSB)
target_compile_options(xtwcore PRIVATE -Wall -Wextra -Wpedantic)
add_executable(xtw5-linux src/main.cpp src/mainwindow.cpp src/hexmodel.cpp)
target_link_libraries(xtw5-linux PRIVATE xtwcore Qt6::Widgets Qt6::Concurrent)
target_compile_options(xtw5-linux PRIVATE -Wall -Wextra -Wpedantic)
add_executable(xtw5-hardware-check EXCLUDE_FROM_ALL tools/hardware_check.cpp)
target_link_libraries(xtw5-hardware-check PRIVATE xtwcore)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test-core tests/test_core.cpp)
  target_link_libraries(test-core PRIVATE xtwcore Qt6::Test)
  add_test(NAME core COMMAND test-core)
  add_executable(test-gui tests/test_gui.cpp src/mainwindow.cpp src/hexmodel.cpp)
  target_link_libraries(test-gui PRIVATE xtwcore Qt6::Widgets Qt6::Concurrent Qt6::Test)
  add_test(NAME gui COMMAND test-gui)
  set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS xtw5-linux RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES resources/xtw5-linux.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/xtw5-linux.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES resources/70-xtw5.rules DESTINATION lib/udev/rules.d)
install(FILES README.md docs/distribution.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
include(cmake/Packaging.cmake)
